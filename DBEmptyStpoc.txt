Oracle HR                      SQL Server
------------------------------------------------
HR.REGIONS          ------->   dbo.REGIONS
HR.COUNTRIES        ------->   dbo.COUNTRIES
HR.LOCATIONS        ------->   dbo.LOCATIONS
HR.DEPARTMENTS      ------->   dbo.DEPARTMENTS
HR.JOBS             ------->   dbo.JOBS
HR.EMPLOYEES        ------->   dbo.EMPLOYEES
HR.JOB_HISTORY      ------->   dbo.JOB_HISTORY
HR.DEPENDENTS       ------->   dbo.DEPENDENTS

CREATE TABLE dbo.REGIONS
(
    REGION_ID     INT PRIMARY KEY,
    REGION_NAME   NVARCHAR(25)
);

CREATE TABLE dbo.COUNTRIES
(
    COUNTRY_ID      CHAR(2) PRIMARY KEY,
    COUNTRY_NAME    NVARCHAR(40),
    REGION_ID       INT
);

CREATE TABLE dbo.DEPARTMENTS
(
    DEPARTMENT_ID     INT PRIMARY KEY,
    DEPARTMENT_NAME   NVARCHAR(30),
    MANAGER_ID        INT,
    LOCATION_ID       INT
);

CREATE TABLE dbo.JOBS
(
    JOB_ID        NVARCHAR(10) PRIMARY KEY,
    JOB_TITLE     NVARCHAR(35),
    MIN_SALARY    DECIMAL(12,2),
    MAX_SALARY    DECIMAL(12,2)
);

CREATE TABLE dbo.EMPLOYEES
(
    EMPLOYEE_ID      INT PRIMARY KEY,
    FIRST_NAME       NVARCHAR(20),
    LAST_NAME        NVARCHAR(25),
    EMAIL            NVARCHAR(25),
    PHONE_NUMBER     NVARCHAR(20),
    HIRE_DATE        DATE,
    JOB_ID           NVARCHAR(10),
    SALARY           DECIMAL(12,2),
    COMMISSION_PCT   DECIMAL(5,2),
    MANAGER_ID       INT,
    DEPARTMENT_ID    INT
);

CREATE TABLE dbo.JOB_HISTORY
(
    EMPLOYEE_ID      INT,
    START_DATE       DATE,
    END_DATE         DATE,
    JOB_ID           NVARCHAR(10),
    DEPARTMENT_ID    INT,

    CONSTRAINT PK_JOB_HISTORY
        PRIMARY KEY
        (
            EMPLOYEE_ID,
            START_DATE
        )
);

CREATE TABLE dbo.DEPENDENTS
(
    DEPENDENT_ID    INT PRIMARY KEY,
    FIRST_NAME      NVARCHAR(50),
    LAST_NAME       NVARCHAR(50),
    RELATIONSHIP    NVARCHAR(30),
    EMPLOYEE_ID     INT
);

CREATE TABLE dbo.LOCATIONS
(
    LOCATION_ID      INT PRIMARY KEY,
    STREET_ADDRESS   NVARCHAR(40),
    POSTAL_CODE      NVARCHAR(12),
    CITY             NVARCHAR(30),
    STATE_PROVINCE   NVARCHAR(25),
    COUNTRY_ID       CHAR(2)
);

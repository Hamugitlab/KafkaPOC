If you want a JDBC Source Connector that reads all tables from a PostgreSQL schema, you don't need table.whitelist.
Next step: PostgreSQL → Kafka → SQL Server

PostgreSQL
    ↓
JDBC Source Connector
    ↓
Kafka Topics (pg_public_*)
    ↓
ksqlDB (optional)
    ↓
JDBC Sink Connector
    ↓
SQL Server



Example connector:

 cat > postgres-source_Schema.json <<'EOF' 
{
  "name": "postgres-source_Schema",
  "config": {
    "name": "postgres-source_Schema",
    "connector.class": "io.confluent.connect.jdbc.JdbcSourceConnector",
    "tasks.max": "1",
    "connection.url": "jdbc:postgresql://postgres:5432/mydb",
    "connection.user": "postgres",
    "connection.password": "********",
    "schema.pattern": "public",
    "mode": "incrementing",
    "incrementing.column.name": "id",
    "timestamp.column.name": "",
    "poll.interval.ms": "5000",
    "topic.prefix": "pg_public_"
  }
}
EOF


curl -X POST \
  -H "Content-Type: application/json" \
  --data @postgres-source_Schema.json \
  http://localhost:8083/connectors
  
  curl http://localhost:8083/connectors/postgres-source_Schema/status

  "name": "postgres-cdc",
  "config": {
    "transforms.unwrap.delete.handling.mode": "rewrite",

    "topic.prefix": "postgres",
    "database.hostname": "postgres",
    "database.port": "5432",
    "database.user": "debezium",
    "database.password": "********",
    "database.dbname": "mydb",
    "plugin.name": "pgoutput",
    "slot.name": "debezium_slot",
    "publication.name": "dbz_publication",
    "publication.autocreate.mode": "disabled"

}

SQL Server Sink Connector



cat > sqlserver-sink.json <<'EOF'
{
  "name": "sqlserver-sink",
  "config": {
    "connector.class": "io.confluent.connect.jdbc.JdbcSinkConnector",
    "tasks.max": "1",

    "topics.regex": "postgres.public.*",

    "connection.url": "jdbc:sqlserver://him-mssql_01:1433;databaseName=himtestdb;encrypt=false;trustServerCertificate=true",
    "connection.user": "sa",
    "connection.password": "hello@123",

    "insert.mode": "upsert",

    "pk.mode": "record_key",
    "pk.fields": "id",

    "delete.enabled": "true",

    "auto.create": "true",
    "auto.evolve": "true",

    "transforms": "unwrap,route",

    "transforms.unwrap.type": "io.debezium.transforms.ExtractNewRecordState",
    "transforms.unwrap.drop.tombstones": "true",
    "transforms.unwrap.delete.handling.mode": "rewrite",

    "transforms.route.type": "org.apache.kafka.connect.transforms.RegexRouter",
    "transforms.route.regex": "postgres\\.public\\.(.*)",
    "transforms.route.replacement": "$1",

    "table.name.format": "${topic}"
  }
}
EOF

curl -X POST \
-H "Content-Type: application/json" \
--data @sqlserver-sink.json \
http://localhost:8083/connectors

 Clean PostgreSQL replication artifacts
 
 
 DROP PUBLICATION IF EXISTS dbz_publication; 
 
 SELECT slot_name
FROM pg_replication_slots;

SELECT pg_drop_replication_slot('debezium_slot');

SELECT pg_terminate_backend(active_pid)
FROM pg_replication_slots
WHERE slot_name='debezium_slot';
